const express = require('express'); const path = require('path'); const app = express(); const PORT = process.env.PORT || 3000; const ANTHROPIC_API_KEY = process.env.ANTHROPIC_API_KEY; const CLAUDE_MODEL = 'claude-sonnet-4-6'; app.use(express.json()); app.use(express.static(path.join(__dirname, 'public'))); // Format-specific editor notes injected into the Vault Instructions // so the script includes direction relevant to how it will be filmed/edited. const FORMAT_EDITOR_NOTES = { 'Talking Head': 'Write for direct-to-camera delivery. No B-roll cues needed; keep lines punchy enough to hold attention on face alone.', 'Voice-Over with B-Roll': 'Write as a voice-over script. After each line or beat, include a bracketed B-ROLL cue suggesting what footage should play under that line.', 'Text-Hook Loop': 'Structure the hook as on-screen text that loops or escalates. Include bracketed ON-SCREEN TEXT cues for the hook and key beats.', 'Mixed Format': 'Blend direct-to-camera lines with voice-over segments. Include bracketed cues marking which lines are on-camera vs. voice-over/B-roll.', 'Green Screen': 'Write for a green screen setup with a background image/video behind the speaker. Include bracketed BACKGROUND cues suggesting what should appear behind the speaker at key moments.', 'Skit Roleplay': 'Write as a short scripted skit with at least two roles or a clear character switch. Label who is speaking at each line.', 'POV': 'Write in second-person POV style ("you walk in and...") as if the viewer is living the scenario. Include bracketed SHOT cues for what the camera sees.', 'Day in the Life Vlog': 'Write as a loose vlog voice-over that follows a real-time sequence of moments. Include bracketed TIMESTAMP/SCENE cues (e.g., Morning, Midday, Evening).', 'Case Study Teardown': 'Write as a breakdown/analysis of a real or example result. Include bracketed cues for when to show screen recordings, screenshots, or data overlays.', 'Watch Me Work ASMR': 'Write minimal spoken lines meant to sit over quiet ambient/ASMR-style work footage. Include bracketed cues for on-screen text carrying the main educational points since dialogue should be sparse.', 'Interview QA': 'Write in a question-and-answer format with a clear interviewer question followed by the client/creator answer for each beat.' }; function buildVaultInstructions(hookCategory, template, videoFormat) { const hookLine = hookCategory && hookCategory !== 'Any AI chooses best' ? `Hook Category: ${hookCategory}. The HOOK must be written strictly in this style. Do not default to a generic hook style.` : 'Hook Category: Any AI chooses best. Select the hook style most likely to stop the scroll for this topic and audience.'; const templateLine = template && template !== 'Any AI chooses best' ? `Script Template: ${template}. The overall structure of the BODY must follow this template's narrative/structural pattern exactly, not a generic structure.` : 'Script Template: Any AI chooses best. Select the structural template best suited to this topic and audience.'; const formatNote = FORMAT_EDITOR_NOTES[videoFormat] || 'Write generally, but include any bracketed production cues that would help an editor understand pacing.'; const formatLine = `Video Format: ${videoFormat}. ${formatNote}`; return `VAULT INSTRUCTIONS (HARD REQUIREMENTS — do not deviate from these): - ${hookLine} - ${templateLine} - ${formatLine} These three selections are non-negotiable creative constraints. Every part of the script (hook, body, and CTA delivery) must reflect them.`; } function buildPrompt(fields) { const { clientName, niche, packageTier, offer, targetAudience, toneOfVoice, scriptTopic, videoLength, callToAction, hookCategory, template, videoFormat, additionalNotes } = fields; const vaultInstructions = buildVaultInstructions(hookCategory, template, videoFormat); return `You are an expert short-form content scriptwriter working for First Class Creators, an agency that writes Instagram Reel scripts for coaches, course creators, and digital service providers. Write one complete short-form video script based on the details below. Return the script formatted with clearly bolded HOOK, BODY, and CTA sections, in that order. No emojis, no dashes. CLIENT DETAILS Client Name: ${clientName} Niche: ${niche} Package Tier: ${packageTier} scripts/month Their Offer: ${offer} Target Audience: ${targetAudience} Tone of Voice: ${toneOfVoice} Video Length: ${videoLength} seconds Script Topic: ${scriptTopic} Call to Action: ${callToAction} Additional Notes: ${additionalNotes || 'None'} ${vaultInstructions} Write the full script now, formatted as: HOOK: [hook lines] BODY: [body lines] CTA: [cta lines]`; } app.post('/api/generate-script', async (req, res) => { try { const { clientName, niche, packageTier, offer, targetAudience, toneOfVoice, scriptTopic, videoLength, callToAction, hookCategory, template, videoFormat, additionalNotes } = req.body || {}; if (!clientName || !niche || !offer || !targetAudience || !toneOfVoice || !scriptTopic || !callToAction) { return res.status(400).json({ error: 'Missing required fields.' }); } if (!ANTHROPIC_API_KEY) { return res.status(500).json({ error: 'Server is missing ANTHROPIC_API_KEY.' }); } const resolvedHookCategory = hookCategory || 'Any AI chooses best'; const resolvedTemplate = template || 'Any AI chooses best'; const resolvedVideoFormat = videoFormat || 'Talking Head'; const prompt = buildPrompt({ clientName, niche, packageTier, offer, targetAudience, toneOfVoice, scriptTopic, videoLength, callToAction, hookCategory: resolvedHookCategory, template: resolvedTemplate, videoFormat: resolvedVideoFormat, additionalNotes }); const anthropicRes = await fetch('https://api.anthropic.com/v1/messages', { method: 'POST', headers: { 'Content-Type': 'application/json', 'x-api-key': ANTHROPIC_API_KEY, 'anthropic-version': '2023-06-01' }, body: JSON.stringify({ model: CLAUDE_MODEL, max_tokens: 1500, messages: [ { role: 'user', content: prompt } ] }) }); if (!anthropicRes.ok) { const errText = await anthropicRes.text(); console.error('Anthropic API error:', errText); return res.status(502).json({ error: 'Script generation failed. Please try again.' }); } const anthropicData = await anthropicRes.json(); const scriptText = (anthropicData.content || []) .map((block) => (block.type === 'text' ? block.text : '')) .filter(Boolean) .join('\n') .trim(); return res.json({ script: scriptText, hookCategory: resolvedHookCategory, template: resolvedTemplate, videoFormat: resolvedVideoFormat }); } catch (err) { console.error('Error generating script:', err); return res.status(500).json({ error: 'Unexpected server error.' }); } }); app.listen(PORT, () => { console.log(`FCC Script Generator running on port ${PORT}`); });
